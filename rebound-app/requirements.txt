# Rebound predictor backend.
#
# The feature code lives in the `rebounding` package, not here -- there is exactly one
# definition of a feature and it is in the ReboundingPrediction repo. Install it from a
# checkout beside this one:
#   pip install -e '../ReboundingPrediction[serve]'
# or straight from git:
#   pip install 'rebounding[serve] @ git+https://github.com/riders994/ReboundingPrediction@<commit>'
#
# Versions below are the ones the shipped bundles were BUILT with (python 3.12.3).
# They are pinned rather than floored because the artifacts are pickled dataclass
# bundles: unpickling imports `rebounding` and reconstructs numpy/pandas objects, so a
# drifting minor version is a runtime failure, not a deprecation warning.

Flask==3.1.3
gunicorn==26.1.0

numpy==2.5.2          # NB: 2.x. The old pin was numpy>=1.24,<2.0 and is incompatible.
pandas==3.0.5
scipy==1.18.0
pyarrow==25.0.1
joblib==1.5.3
lightgbm==4.7.0

# Required even though nothing served here is an sklearn estimator: the rebounder's
# BoostedSoftmax wraps LGBMRegressor and unpickling pulls sklearn in.
scikit-learn==1.9.0

# The movement model (the animation). Optional and fail-soft -- without it the
# rebounder still works and every player simply stays where he was placed -- but the
# animation is the demo's whole interaction, so it belongs in the deploy.
# Install the CPU wheel or pip drags in ~2.5 GB of CUDA for a 800k-parameter model:
#   pip install torch --index-url https://download.pytorch.org/whl/cpu
torch==2.13.0+cpu

# Only needed if REBOUND_CORS_ORIGINS is set. nginx serves the site and the API from
# the same origin in production, so it normally is not.
# flask-cors>=4.0

# Prediction logging (telemetry.py). Only does anything when REBOUND_DATABASE_URL is
# set -- with no URL the module never imports it at all, and with a URL but no driver
# logging degrades to a one-line reason on /healthz. Installed by default anyway so that
# turning logging on is an environment change rather than a redeploy.
#
# A floor, not a pin, unlike everything above: those are pinned because unpickling the
# bundles reconstructs numpy/pandas objects and a drifting minor version is a runtime
# failure. Nothing here is unpickled, so psycopg only has to satisfy its own API.
# `[binary]` ships the libpq wheel; drop the extra if you would rather link the system
# libpq (`sudo apt install libpq5`).
psycopg[binary]>=3.2,<4

# TensorFlow is gone: posnn.h5 was retrained as MovementModel.pkl and nothing loads
# Keras any more. msd.pkl has no successor -- the normalisation constants are buffers
# inside the bundle, where they cannot drift from the weights.
